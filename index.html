<script>

  // Зачеркивание чекбоксов

  document.querySelectorAll('.checklist-item input').forEach((checkbox, index) => {

    // Восстановление состояния из localStorage

    const savedState = localStorage.getItem('chk_' + index);

    if (savedState === 'true') {

      checkbox.checked = true;

      checkbox.closest('.checklist-item').classList.add('done');

    }


    checkbox.addEventListener('change', (e) => {

      const item = e.target.closest('.checklist-item');

      item.classList.toggle('done', e.target.checked);

      localStorage.setItem('chk_' + index, e.target.checked);

    });

  });


  // Автосохранение текстовых полей шапки

  ['trainee-name', 'start-date', 'mentors-list'].forEach(id => {

    const el = document.getElementById(id);

    if (localStorage.getItem(id)) el.value = localStorage.getItem(id);

    el.addEventListener('input', () => localStorage.setItem(id, el.value));

  });


  // Автосохранение таблицы передачи смены

  const tableBody = document.getElementById('handover-body');

  if (localStorage.getItem('table_data')) {

    tableBody.innerHTML = localStorage.getItem('table_data');

  }


  tableBody.addEventListener('input', () => {

    localStorage.setItem('table_data', tableBody.innerHTML);

  });
